🟢 Integração com Google Sheets

1. Crie uma planilha no Google Sheets

sheets.google.com → Nova planilha em branco.

2. Abra o Apps Script

Extensões → Apps Script. Apague tudo e cole:

// ── Cole TODO este código no Apps Script ── // doGet: necessário para o Google validar a URL // e para o botão "Buscar da Planilha" funcionar function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('CRM') || ss.getActiveSheet(); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var contacts = rows.slice(1).map(function(r) { var obj = {}; headers.forEach(function(h, i) { obj[String(h).toLowerCase().replace(/\s/g,'_')] = r[i]; }); // Normaliza campos principais return { nome: obj.nome || obj.name || '', phone: String(obj.telefone || obj.phone || '').replace(/\D/g,''), estado: obj.estado || '', status: obj.status || obj.situação || '', lastContact: obj.último_contato || obj.last_contact || '', nextContact: obj.próximo_contato || obj.next_contact || '', sentAt: obj.enviado_em || '', notes: obj.anotações || obj.notes || '' }; }).filter(function(c){ return c.phone; }); return ContentService .createTextOutput(JSON.stringify({ok:true, contacts:contacts})) .setMimeType(ContentService.MimeType.JSON); } // doPost: recebe dados do ADV SHOT e salva na aba CRM function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('CRM'); if (!sheet) { sheet = ss.insertSheet('CRM'); sheet.appendRow(['Nome','Telefone','Estado','Status','Temperatura', 'Próximo Contato','Último Contato','Link WA','Anotações']); } var statusMap = { 'pendente':'Aguardando','respondeu':'Respondeu','interessado':'Interessado', 'retornar':'Retornar','nao_quer':'Não quer','sem_viabilidade':'Sem viabilidade', 'fechado':'Fechado','errado':'Nº errado' }; var tempMap = { 'interessado':'Quente','respondeu':'Morno','retornar':'Morno', 'pendente':'Frio','nao_quer':'Frio','sem_viabilidade':'Frio', 'fechado':'Fechado','errado':'Frio' }; var lastRow = sheet.getLastRow(); var phoneIndex = {}; if (lastRow > 1) { sheet.getRange(2,2,lastRow-1,1).getValues().forEach(function(r,i){ var p=String(r[0]).replace(/\D/g,''); if(p) phoneIndex[p]=i+2; }); } var added=0, updated=0; (data.contacts||[]).forEach(function(c){ var phone=String(c.phone||'').replace(/\D/g,''); var nome=c.nome||'', sit=statusMap[c.status]||c.status||''; var temp=tempMap[c.status]||'', prox=c.nextContact||''; var ult=c.lastContact||'', link=phone?'https://wa.me/'+phone:''; var notes=c.notes||'', estado=c.estado||''; if(phoneIndex[phone]){ var row=phoneIndex[phone]; sheet.getRange(row,1).setValue(nome); sheet.getRange(row,3).setValue(estado); sheet.getRange(row,4).setValue(sit); sheet.getRange(row,5).setValue(temp); if(prox)sheet.getRange(row,6).setValue(prox); if(ult)sheet.getRange(row,7).setValue(ult); sheet.getRange(row,8).setValue(link); if(notes)sheet.getRange(row,9).setValue(notes); updated++; } else { var nr=sheet.getLastRow()+1; sheet.getRange(nr,1).setValue(nome); sheet.getRange(nr,2).setValue(phone); sheet.getRange(nr,3).setValue(estado); sheet.getRange(nr,4).setValue(sit); sheet.getRange(nr,5).setValue(temp); sheet.getRange(nr,6).setValue(prox); sheet.getRange(nr,7).setValue(ult); sheet.getRange(nr,8).setValue(link); sheet.getRange(nr,9).setValue(notes); phoneIndex[phone]=nr; added++; } }); return ContentService .createTextOutput(JSON.stringify({ok:true,added:added,updated:updated})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ok:false,error:err.message})) .setMimeType(ContentService.MimeType.JSON); } }

3. Implante como Web App

Implantar → Nova implantação → Tipo: App da Web → Executar como: Eu → Acesso: Qualquer pessoa → Implantar. Copie a URL.

4. Cole a URL na Home

Cole no campo Google Sheets e clique Sincronizar.

✅ Sobre limites: Google Sheets suporta 10 milhões de células. Para 5.000 leads × 10 campos = 50.000 células — você poderia ter 200× mais leads e ainda estaria dentro do limite. Totalmente gratuito.
